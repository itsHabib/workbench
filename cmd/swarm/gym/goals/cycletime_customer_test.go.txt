package customer

import (
	"os"
	"strings"
	"testing"
)

// The customer's acceptance test. It knows nothing about the team's
// packages; it checks what the change asked for in ways any layout meets.
func TestCustomerCSVExampleExists(t *testing.T) {
	entries, err := os.ReadDir("../testdata")
	if err != nil {
		t.Fatalf("no testdata directory at the repository root: %v", err)
	}
	for _, e := range entries {
		if strings.HasSuffix(e.Name(), ".csv") {
			return
		}
	}
	t.Fatal("no .csv example job under testdata/")
}

func TestCustomerREADMEShowsCSV(t *testing.T) {
	data, err := os.ReadFile("../README.md")
	if err != nil {
		t.Fatal(err)
	}
	text := strings.ToLower(string(data))
	if !strings.Contains(text, ".csv") {
		t.Fatal("README worked example does not use a CSV job")
	}
	if !strings.Contains(text, "setup") {
		t.Fatal("README does not mention setup time")
	}
}
